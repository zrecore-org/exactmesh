# Test suites (Catch2 v3, fetched and hash-checked by third_party/fetch.sh catch2):
#   unit_*          fast suite, label "fast"                    ctest -L fast
#   unit_*_mut_*    planted mutants of the code under test; each MUST make its suite fail (WILL_FAIL), label "mutant"
#   acceptance      BDD scenarios through the pinned, patched Geogram pipeline, label "acceptance"; needs EXACTMESH_FIDELITY2=<path of geogram_fidelity2>
#                   (built by ci/geogram_xplat.sh build-chain + build-fidelity2); never part of the fast suite
#   gpu_*           orient filter and LBVH against their CPU references, label "gpu"; only with -DEXACTMESH_GPU=ON
set(CATCH2_DIR "${EXACTMESH_DEPS}/Catch2-3.16.1")
if(NOT EXISTS "${CATCH2_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Catch2 not found in ${EXACTMESH_DEPS}: run third_party/fetch.sh catch2 (or configure with -DEXACTMESH_TESTS=OFF)")
endif()
set(CATCH_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(CATCH_INSTALL_EXTRAS OFF CACHE BOOL "" FORCE)
set(CATCH_DEVELOPMENT_BUILD OFF CACHE BOOL "" FORCE)
add_subdirectory("${CATCH2_DIR}" "${CMAKE_BINARY_DIR}/catch2")

file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test_work")
function(exactmesh_test name)
  add_executable(${name} ${ARGN})
  target_include_directories(${name} PRIVATE "${BOOST_ROOT_DIR}" "${CMAKE_SOURCE_DIR}/harness" "${CMAKE_SOURCE_DIR}/lib/geogram/mp21b" "${CMAKE_SOURCE_DIR}/tests/common" "${CMAKE_SOURCE_DIR}/gpu/shaders")
  target_compile_definitions(${name} PRIVATE EXACTMESH_TEST_TMP="${CMAKE_BINARY_DIR}/test_work/")
  target_link_libraries(${name} PRIVATE Catch2::Catch2WithMain)
endfunction()

# ---- core units: exact arithmetic, Shewchuk predicates vs exact, soft binary64, topology, point in solid
exactmesh_test(unit_core unit/test_exact.cpp unit/test_predicates.cpp unit/test_soft_f64.cpp unit/test_topology_pip.cpp)

add_test(NAME unit_core COMMAND unit_core "[unit]")
set_tests_properties(unit_core PROPERTIES LABELS "fast")

# ---- scaling and re-centring rules of the pipeline, taken verbatim from lib/geogram/mp24b/geogram_fidelity2.cpp (no Geogram needed)
set(F2_SRC "${CMAKE_SOURCE_DIR}/lib/geogram/mp24b/geogram_fidelity2.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${F2_SRC}")
file(READ "${F2_SRC}" F2_TEXT)
string(FIND "${F2_TEXT}" "#ifndef T_EXP" RULES_BEGIN)
string(FIND "${F2_TEXT}" "}  // namespace" RULES_END)
if(RULES_BEGIN EQUAL -1 OR RULES_END EQUAL -1)
  message(FATAL_ERROR "cannot find the rules section in geogram_fidelity2.cpp")
endif()
math(EXPR RULES_LEN "${RULES_END} + 15 - ${RULES_BEGIN}")
string(SUBSTRING "${F2_TEXT}" ${RULES_BEGIN} ${RULES_LEN} F2_RULES)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/pipeline_rules.h"
"// Generated by tests/CMakeLists.txt from lib/geogram/mp24b/geogram_fidelity2.cpp: the scale / re-centre / refuse rules, unchanged.\n#pragma once\n#include \"mesh_io.h\"\n#include <cfloat>\n#include <cmath>\n#include <algorithm>\n#include <string>\n#include <cstdio>\n#include <cstdint>\n${F2_RULES}\n")
exactmesh_test(unit_rules unit/test_rules.cpp)
target_include_directories(unit_rules PRIVATE "${CMAKE_BINARY_DIR}/generated")
add_test(NAME unit_rules COMMAND unit_rules "[unit]")
set_tests_properties(unit_rules PROPERTIES LABELS "fast")
foreach(M NOSCALE SCALE19 C0 TLARGE)
  exactmesh_test(unit_rules_mut_${M} unit/test_rules.cpp)
  target_include_directories(unit_rules_mut_${M} PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(unit_rules_mut_${M} PRIVATE MUT_${M})
  add_test(NAME unit_rules_mut_${M} COMMAND unit_rules_mut_${M} "[unit]")
  set_tests_properties(unit_rules_mut_${M} PROPERTIES LABELS "mutant" WILL_FAIL TRUE)
endforeach()
